## Project tree

```shell
project
├── CMakeLists.txt
└── main.cpp
```

## CMakeLists.txt

```cmake title="CMakeLists.txt"
cmake_minimum_required(VERSION 3.15)

set(PROJECT_NAME "example")
project(${PROJECT_NAME} LANGUAGES CXX)

set(TARGET_NAME "${PROJECT_NAME}_main")

set(SOURCE
    "main.cpp"
)

add_executable(${TARGET_NAME} ${SOURCE})
```
