is_installed fzf && {
    eval "$(fzf --zsh)"
    # fzf
    export FZF_DEFAULT_OPTS='--height 60% --layout=reverse --border'
    is_installed fd && {
        export FZF_DEFAULT_COMMAND='fd --type f --follow' # --hidden --exclude .git
        export FZF_CTRL_T_COMMAND="$FZF_DEFAULT_COMMAND"
        export FZF_CTRL_T_OPTS="--preview 'cat {}'"
    }
}
